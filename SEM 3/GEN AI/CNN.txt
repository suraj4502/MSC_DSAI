# Step 3: Extract the downloaded zip file
import zipfile
zip_ref = zipfile.ZipFile('/content/pediatric-pneumonia-chest-xray.zip', 'r')
zip_ref.extractall('/content')
zip_ref.close()

# Step 4: Import necessary libraries for TensorFlow and Keras
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, Conv2D, MaxPool2D, Flatten

# Step 5: Load image datasets from extracted folders
tr_data = tf.keras.utils.image_dataset_from_directory(
    directory='/content/Pediatric Chest X-ray Pneumonia/train',
    labels='inferred',
    label_mode='int',
    batch_size=32,
    image_size=(256, 256)
)

test_data = tf.keras.utils.image_dataset_from_directory(
    directory='/content/Pediatric Chest X-ray Pneumonia/test',
    labels='inferred',
    label_mode='int',
    batch_size=32,
    image_size=(256, 256)
)

# Step 6: Normalize images
def normalize(image, label):
    image = tf.cast(image / 255.0, tf.float32)
    return image, label

tr_data = tr_data.map(normalize)
test_data = test_data.map(normalize)

# Step 7: Define Convolutional Neural Network model
model = Sequential()
model.add(Conv2D(32, (3, 3), activation='relu', input_shape=(256, 256, 3)))
model.add(MaxPool2D((2, 2), strides=2, padding='valid'))
model.add(Conv2D(64, (3, 3), activation='relu'))
model.add(MaxPool2D((2, 2), strides=2, padding='valid'))
model.add(Conv2D(128, (3, 3), activation='relu'))
model.add(MaxPool2D((2, 2), strides=2, padding='valid'))
model.add(Flatten())
model.add(Dense(128, activation='relu'))
model.add(Dense(64, activation='relu'))
model.add(Dense(1, activation='sigmoid'))

model.summary()

# Step 8: Compile and train the model
model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])
model.fit(tr_data, epochs=10, validation_data=test_data)

# Step 9: Load and preprocess a single test image for prediction
from tensorflow.keras.utils import load_img
import numpy as np
import matplotlib.pyplot as plt

test_img = load_img('/content/Pediatric Chest X-ray Pneumonia/test/NORMAL/IM-0001-0001.jpeg', target_size=(256, 256))
plt.imshow(test_img)
plt.axis('off')
plt.show()

test_img = np.array(test_img)
test_img = test_img / 255.0  # Normalize
test_img = test_img.reshape((1, 256, 256, 3))  # Reshape for model input

# Step 10: Make prediction on the image
prediction = model.predict(test_img)
if prediction >= 0.5:
    print('Pneumonia')
else:
    print('Normal')
