## EXTERNAL
CREATE EXTERNAL TABLE IF NOT EXISTS sample_table (
  id INT,
  name STRING,
  age INT,
  city STRING
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/user/hive/external/sample_data'
TBLPROPERTIES ("skip.header.line.count"="1");

## PARTITIONED + external
CREATE EXTERNAL TABLE emps_external_partitioned (
    id INT, 
    name STRING, 
    age INT
) 
PARTITIONED BY (city STRING) 
ROW FORMAT DELIMITED 
FIELDS TERMINATED BY ',' 
STORED AS TEXTFILE 
LOCATION '/msc/external_table_data' 
TBLPROPERTIES ("skip.header.line.count"="1");

## bucketed
CREATE TABLE employees_bucketed (
    id INT,
    name STRING,
    age INT,
    city STRING
)
CLUSTERED BY (id) INTO 4 BUCKETS
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE;
